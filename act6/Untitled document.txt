import heapq


def path_finder(area):
    if not area:
        return 0
    
    grid = area.split('\n')
    N = len(grid)
    
    if N == 1:
        return 0
    
    dist = [[float('inf')] * N for _ in range(N)]
    dist[0][0] = 0
    heap = [(0, 0, 0)]
    
    directions = [(0,1),(0,-1),(1,0),(-1,0)]
    
    while heap:
        cost, r, c = heapq.heappop(heap)
        
        if r == N-1 and c == N-1:
            return cost
        
        if cost > dist[r][c]:
            continue
        
        for dr, dc in directions:
            nr, nc = r + dr, c + dc
            if 0 <= nr < N and 0 <= nc < N:
                new_cost = cost + abs(int(grid[nr][nc]) - int(grid[r][c]))
                if new_cost < dist[nr][nc]:
                    dist[nr][nc] = new_cost
                    heapq.heappush(heap, (new_cost, nr, nc))
    
    return dist[N-1][N-1]
